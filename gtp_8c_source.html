32 * @param line string to process

34void gtp_preprocess(char *line)
35{
36 char *s;
37
38 for (s = line; *line; ++line) {
39 if (*line == '\t') *line = ' ';
40 if (*line == '#') break;
41 if (*line < 32 && *line != '\n') continue;
42
43 *s++ = *line;
44 }
45 *s = '\0';
46}

79 *color = WHITE;
